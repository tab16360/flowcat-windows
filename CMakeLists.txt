cmake_minimum_required(VERSION 3.24)
project(FlowCatWindows VERSION 0.2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /EHsc)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(FlowCatShared STATIC src/common.cpp src/serialization.cpp src/network.cpp src/ipc.cpp)
target_link_libraries(FlowCatShared PUBLIC advapi32 shell32 ole32 runtimeobject windowsapp)
add_executable(FlowCat WIN32 src/main.cpp src/client.cpp src/settings.cpp src/update.cpp app.rc)
target_link_options(FlowCat PRIVATE /MANIFEST:NO)
target_link_libraries(FlowCat PRIVATE FlowCatShared wlanapi shlwapi comctl32 d2d1 dwrite dwmapi psapi comdlg32 uxtheme winhttp bcrypt version)
add_executable(FlowCatService src/service.cpp src/engine.cpp src/collector.cpp src/store.cpp src/health.cpp service.rc)
target_link_libraries(FlowCatService PRIVATE FlowCatShared tdh iphlpapi ws2_32)
add_executable(FlowCatCli tests/cli.cpp src/update.cpp)
target_link_libraries(FlowCatCli PRIVATE FlowCatShared winhttp bcrypt version)
add_executable(FlowCatCoreTests tests/core.cpp)
add_executable(FlowCatStoreTests tests/store.cpp src/store.cpp)
target_link_libraries(FlowCatStoreTests PRIVATE shell32 ole32)
add_executable(FlowCatUpdateTests tests/update.cpp src/update.cpp)
target_link_libraries(FlowCatUpdateTests PRIVATE FlowCatShared winhttp bcrypt version)
add_executable(FlowCatUiCapture WIN32 tests/ui_capture.cpp)
target_link_libraries(FlowCatUiCapture PRIVATE ole32 windowscodecs shell32)
enable_testing()
add_test(NAME core COMMAND FlowCatCoreTests)
add_test(NAME persistence COMMAND FlowCatStoreTests)
add_test(NAME update COMMAND FlowCatUpdateTests)
add_executable(FlowCatFailService EXCLUDE_FROM_ALL tests/fail_service.cpp)
